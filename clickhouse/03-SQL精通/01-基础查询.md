# 第 3 章：SQL 精通

## 3.1 SELECT 基础

### 3.1.1 基本语法

```sql
SELECT [DISTINCT] expr_list
[FROM [db.]table | (subquery) | table_function]
[PREWHERE expr]
[WHERE expr]
[GROUP BY expr]
[HAVING expr]
[ORDER BY expr [ASC|DESC] [WITH FILL]]
[LIMIT [offset,] n]
[SETTINGS ...]
```

### 3.1.2 SELECT 子句

```sql
-- 基础查询
SELECT 1;
SELECT 'Hello' AS greeting, now() AS ts;

-- 数学运算
SELECT
    amount,
    amount * 0.9 AS discounted,
    round(amount / 100, 2) AS amount_yuan
FROM orders;

-- 条件判断
SELECT
    order_id,
    multiIf(
        amount >= 1000, '大单',
        amount >= 100,  '中单',
        '小单'
    ) AS order_size
FROM orders;
```

### 3.1.3 WHERE 子句

```sql
-- 基础过滤
SELECT * FROM orders WHERE amount > 1000;

-- 多条件
SELECT * FROM orders
WHERE order_date = '2024-01-01'
  AND amount > 100
  AND status IN ('paid', 'shipped');

-- 范围查询
SELECT * FROM orders
WHERE order_date BETWEEN '2024-01-01' AND '2024-01-31'
  AND amount >= 100;

-- 模糊查询
SELECT * FROM users
WHERE phone LIKE '138%';  -- ⚠️ 不走索引，全表扫描

-- ✅ 优化：使用函数
SELECT * FROM users
WHERE startsWith(phone, '138');
```

### 3.1.4 PREWHERE 优化

```sql
-- PREWHERE 是 ClickHouse 特有优化
-- 只读取需要的列到内存，先过滤，再读剩余列

-- 示例：events 表有 100 列，但查询只关心 event_type
-- 普通 WHERE：读所有 100 列 → 过滤
-- PREWHERE：先读 event_type → 过滤 → 再读其他需要的列

-- ✅ 推荐：让 ClickHouse 自动判断
-- 手动指定场景：复杂条件
SELECT count()
FROM events
PREWHERE event_date = '2024-01-01'  -- 先用这一列过滤
WHERE user_id > 1000;                -- 再用第二列
```

**性能对比**：
```sql
-- 1 亿行表，event_date 过滤后剩 1% 数据

-- 普通 WHERE
SELECT count() FROM events WHERE event_date = '2024-01-01';
-- 耗时：0.8s

-- PREWHERE
SELECT count() FROM events PREWHERE event_date = '2024-01-01';
-- 耗时：0.3s （快 2-3 倍）
```

## 3.2 GROUP BY 聚合

### 3.2.1 基础聚合

```sql
-- 统计每个城市的订单数
SELECT
    city,
    count() AS order_count,
    sum(amount) AS total_amount,
    avg(amount) AS avg_amount,
    min(amount) AS min_amount,
    max(amount) AS max_amount
FROM orders
WHERE order_date = '2024-01-01'
GROUP BY city
ORDER BY order_count DESC
LIMIT 10;
```

### 3.2.2 高级聚合函数

```sql
-- 1. 去重统计
SELECT
    uniq(user_id) AS uv_estimate,     -- 估算（O(1) 内存）
    uniqExact(user_id) AS uv_exact,   -- 精确（O(N) 内存）
    uniqHLL12(user_id) AS uv_hll      -- HLL 算法
FROM events;

-- 2. 分位数
SELECT
    quantile(0.5)(latency_ms) AS p50,
    quantile(0.95)(latency_ms) AS p95,
    quantile(0.99)(latency_ms) AS p99,
    quantiles(0.5, 0.9, 0.95, 0.99)(latency_ms) AS percentiles
FROM request_logs;

-- 3. 数组聚合
SELECT
    groupArray(product_id) AS products,
    groupUniqArray(product_id) AS unique_products
FROM order_items
GROUP BY order_id;

-- 4. 状态聚合（用于合并）
SELECT
    --argMin(user_id, login_time) AS first_login_user,
    --argMax(user_id, login_time) AS last_login_user
FROM user_events;
```

### 3.2.3 GROUP BY 优化

```sql
-- ❌ 错误：GROUP BY 太多列
SELECT
    user_id,
    event_date,
    event_type,
    city,
    province,
    device,
    os,
    browser,
    count()
FROM events
GROUP BY user_id, event_date, event_type, city, province, device, os, browser;
-- 内存爆炸

-- ✅ 优化：用 -State 函数 + 物化视图
-- 第一步：聚合状态
CREATE MATERIALIZED VIEW events_state_mv
ENGINE = SummingMergeTree
ORDER BY (user_id, event_date)
AS SELECT
    user_id,
    event_date,
    countState() AS cnt
FROM events
GROUP BY user_id, event_date;

-- 第二步：直接查询状态
SELECT
    user_id,
    event_date,
    countMerge(cnt) AS total
FROM events_state_mv
GROUP BY user_id, event_date;
```

## 3.3 JOIN 详解

### 3.3.1 基础 JOIN

```sql
-- ClickHouse 不擅长小表驱动大表的传统 JOIN
-- 但支持所有标准 JOIN 语义

-- INNER JOIN
SELECT
    o.order_id,
    o.amount,
    u.username,
    u.city
FROM orders o
INNER JOIN users u ON o.user_id = u.user_id
WHERE o.order_date = '2024-01-01';

-- LEFT JOIN
SELECT
    u.user_id,
    u.username,
    count(o.order_id) AS order_count
FROM users u
LEFT JOIN orders o ON u.user_id = o.user_id
GROUP BY u.user_id, u.username;

-- 字典 JOIN（ClickHouse 特有优化）
SELECT
    o.order_id,
    o.user_id,
    dictGet('users_dict', 'city', o.user_id) AS city
FROM orders o;
```

### 3.3.2 JOIN 性能对比

```sql
-- 测试：1 亿行 orders JOIN 100 万行 users

-- 方案 1：普通 JOIN
SELECT count()
FROM orders o
INNER JOIN users u ON o.user_id = u.user_id;
-- 耗时：5-10s

-- 方案 2：IN 子查询（小表驱动）
SELECT count()
FROM orders
WHERE user_id IN (SELECT user_id FROM users);
-- 耗时：1-2s （但需要 users 全部装入内存）

-- 方案 3：字典（极致性能）
CREATE DICTIONARY users_dict
(
    user_id UInt64,
    username String,
    city String
)
PRIMARY KEY user_id
SOURCE(CLICKHOUSE(DB 'dwd' TABLE 'users'))
LIFETIME(MIN 300 MAX 600)
LAYOUT(HASHED());

SELECT
    order_id,
    dictGet('users_dict', 'city', user_id) AS city
FROM orders;
-- 耗时：0.3s （20-30 倍提升）

-- 方案 4：JOIN + 优化（join_use_nulls + 内存配置）
SETTINGS
  join_algorithm = 'hash',        -- 强制哈希 join
  max_bytes_in_join = 1000000000; -- 1GB 内存

SELECT count()
FROM orders o
INNER JOIN users u ON o.user_id = u.user_id;
-- 耗时：2-3s
```

### 3.3.3 ASOF JOIN（最接近的连接）

```sql
-- 场景：股票交易表与价格表 join
-- 找到每笔交易发生时的最新价格

CREATE TABLE trades (ts DateTime, symbol String, price Float64) ENGINE = MergeTree ORDER BY ts;
CREATE TABLE quotes (ts DateTime, symbol String, bid Float64) ENGINE = MergeTree ORDER BY ts;

-- 找到每笔交易对应的最近一次报价
SELECT
    t.ts,
    t.symbol,
    t.price,
    q.bid
FROM trades t
ASOF LEFT JOIN quotes q
    ON t.symbol = q.symbol
   AND t.ts >= q.ts
-- 返回：每笔交易发生前的最后一次报价
```

## 3.4 窗口函数

```sql
-- ClickHouse 21.x+ 完整支持窗口函数

-- ROW_NUMBER
SELECT
    user_id,
    order_date,
    amount,
    row_number() OVER (PARTITION BY user_id ORDER BY order_date) AS order_seq
FROM orders;

-- RANK
SELECT
    user_id,
    amount,
    rank() OVER (PARTITION BY user_id ORDER BY amount DESC) AS amount_rank
FROM orders;

-- 累计求和
SELECT
    order_date,
    amount,
    sum(amount) OVER (ORDER BY order_date) AS cumulative_amount
FROM orders
WHERE user_id = 1001;

-- 移动平均
SELECT
    event_date,
    daily_amount,
    avg(daily_amount) OVER (
        ORDER BY event_date
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ) AS ma7
FROM (
    SELECT event_date, sum(amount) AS daily_amount
    FROM orders
    WHERE user_id = 1001
    GROUP BY event_date
);
```

## 3.5 LIMIT 与采样

```sql
-- LIMIT 基础
SELECT * FROM orders ORDER BY amount DESC LIMIT 10;  -- Top 10

-- LIMIT BY（每个分组的 Top N）
-- 每个用户最近 5 笔订单
SELECT
    user_id,
    order_id,
    order_date,
    amount
FROM orders
ORDER BY
    user_id ASC,
    order_date DESC
LIMIT 5 BY user_id;

-- 采样查询
SELECT * FROM events SAMPLE 0.1;  -- 10% 采样
SELECT * FROM events SAMPLE 1000000;  -- 采样 100 万行
SELECT * FROM events SAMPLE 1/10;  -- 1/10 采样

-- 采样 + WHERE
SELECT count()
FROM events SAMPLE 0.01
WHERE event_date = '2024-01-01';
-- 估算全表结果：count() * 100
```

## 3.7 WITH 子句（CTE）

```sql
-- 公共表表达式
WITH
    toDate('2024-01-01') AS start_date,
    toDate('2024-01-31') AS end_date,
    active_users AS (
        SELECT DISTINCT user_id
        FROM events
        WHERE event_date BETWEEN start_date AND end_date
    )
SELECT
    count() AS active_user_count,
    count() / dateDiff('day', start_date, end_date) AS avg_daily_active
FROM active_users;

-- 递归 CTE（24.x+）
WITH RECURSIVE
    hierarchy AS (
        SELECT id, parent_id, name, 1 AS level
        FROM categories
        WHERE parent_id = 0
        UNION ALL
        SELECT c.id, c.parent_id, c.name, h.level + 1
        FROM categories c
        JOIN hierarchy h ON c.parent_id = h.id
    )
SELECT * FROM hierarchy;
```

## 3.8 实战：复杂业务查询

### 案例 1：用户留存分析

```sql
-- 7 日留存：新用户在第 N 天还活跃的比例
WITH
    new_users AS (
        SELECT user_id, min(event_date) AS register_date
        FROM events
        WHERE event_type = 'register'
        GROUP BY user_id
    ),
    user_activity AS (
        SELECT DISTINCT user_id, event_date
        FROM events
    )
SELECT
    register_date,
    count() AS new_user_count,
    -- 留存率
    countIf(register_date + 1 IN (
        SELECT event_date FROM user_activity
        WHERE user_id = n.user_id
    )) / count() AS d1_retention,
    -- 7 日留存
    countIf(register_date + 7 IN (
        SELECT event_date FROM user_activity
        WHERE user_id = n.user_id
    )) / count() AS d7_retention
FROM new_users n
WHERE register_date BETWEEN '2024-01-01' AND '2024-01-31'
GROUP BY register_date
ORDER BY register_date;
```

### 案例 2：漏斗分析

```sql
-- 电商转化漏斗：浏览 → 加购 → 支付 → 完成
SELECT
    level,
    count() AS user_count,
    round(count() * 100.0 / first_value(count()) OVER (ORDER BY level), 2) AS conversion_rate
FROM (
    SELECT
        user_id,
        max(CASE WHEN event_type = 'view' THEN 1 ELSE 0 END) AS level_1,
        max(CASE WHEN event_type = 'add_to_cart' THEN 1 ELSE 0 END) AS level_2,
        max(CASE WHEN event_type = 'pay' THEN 1 ELSE 0 END) AS level_3,
        max(CASE WHEN event_type = 'complete' THEN 1 ELSE 0 END) AS level_4,
        -- 最高漏斗层级
        arrayJoin(
            arrayFilter(x -> x = 1, [
                max(CASE WHEN event_type = 'view' THEN 1 ELSE 0 END),
                max(CASE WHEN event_type = 'add_to_cart' THEN 1 ELSE 0 END),
                max(CASE WHEN event_type = 'pay' THEN 1 ELSE 0 END),
                max(CASE WHEN event_type = 'complete' THEN 1 ELSE 0 END)
            ])
        ) AS level
    FROM events
    WHERE event_date = '2024-01-01'
    GROUP BY user_id
)
GROUP BY level
ORDER BY level;
```
