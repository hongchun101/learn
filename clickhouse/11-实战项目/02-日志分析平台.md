### 11.2.2 数据接入（多源）

```sql
-- ========================================
-- Kafka 原始日志表
-- ========================================
CREATE TABLE ods.app_logs_kafka
(
    level String,
    service String,
    host String,
    trace_id String,
    span_id String,
    message String,
    user_id UInt64 DEFAULT 0,
    request_uri String DEFAULT '',
    method String DEFAULT '',
    status_code UInt16 DEFAULT 0,
    latency_ms UInt32 DEFAULT 0,
    size_bytes UInt64 DEFAULT 0,
    log_time DateTime DEFAULT now()
) ENGINE = Kafka
SETTINGS
    kafka_broker_list = 'kafka1:9092,kafka2:9092,kafka3:9092',
    kafka_topic_list = 'app_logs',
    kafka_group_name = 'log_consumer_1',
    kafka_format = 'JSONEachRow',
    kafka_handle_error_mode = 'stream',
    kafka_num_consumers = 4,
    kafka_thread_per_consumer = 2,
    kafka_max_block_size = 1048576;

-- ========================================
-- 存储到 MergeTree
-- ========================================
CREATE TABLE ods.app_logs
(
    log_date Date,
    log_time DateTime,
    log_time_ms DateTime64(3) MATERIALIZED toDateTime64(log_time, 3),
    level LowCardinality(String),
    service LowCardinality(String),
    host String,
    trace_id String,
    span_id String,
    message String,
    user_id UInt64,
    request_uri String,
    method LowCardinality(String),
    status_code UInt16,
    latency_ms UInt32,
    size_bytes UInt64,
    INDEX idx_trace trace_id TYPE bloom_filter(0.01) GRANULARITY 4,
    INDEX idx_message message TYPE tokenbf_v1(25600, 3, 0) GRANULARITY 4,
    INDEX idx_uri request_uri TYPE tokenbf_v1(25600, 3, 0) GRANULARITY 4
) ENGINE = MergeTree
PARTITION BY toYYYYMMDD(log_date)
ORDER BY (level, service, log_time)
TTL log_date + INTERVAL 30 DAY;

CREATE MATERIALIZED VIEW app_logs_mv TO ods.app_logs AS
SELECT
    toDate(log_time) AS log_date,
    log_time,
    level,
    service,
    host,
    trace_id,
    span_id,
    message,
    user_id,
    request_uri,
    method,
    status_code,
    latency_ms,
    size_bytes
FROM ods.app_logs_kafka;
```

### 11.2.3 错误率告警

```sql
-- ========================================
-- 每分钟错误率
-- ========================================
CREATE TABLE dws.error_rate_1min
(
    minute DateTime,
    service LowCardinality(String),
    error_count UInt64,
    total_count UInt64,
    error_rate_state AggregateFunction(quantile, Float64)
) ENGINE = AggregatingMergeTree
ORDER BY (service, minute);

CREATE MATERIALIZED VIEW error_rate_mv TO dws.error_rate_1min AS
SELECT
    toStartOfMinute(log_time) AS minute,
    service,
    countIf(level = 'ERROR') AS error_count,
    count() AS total_count,
    quantileState(toFloat64(countIf(level = 'ERROR')) / count()) AS error_rate_state
FROM ods.app_logs
WHERE log_time >= now() - INTERVAL 1 DAY
GROUP BY minute, service;

-- 实时查询错误率
SELECT
    service,
    sum(error_count) AS errors,
    sum(total_count) AS total,
    round(errors * 100.0 / total, 2) AS error_rate_pct
FROM dws.error_rate_1min
WHERE minute >= now() - INTERVAL 5 MINUTE
GROUP BY service
HAVING error_rate_pct > 1
ORDER BY error_rate_pct DESC;
```

### 11.2.4 慢查询分析

```sql
-- ========================================
-- 慢请求 Top N
-- ========================================
CREATE TABLE dws.slow_requests
(
    window_start DateTime,
    request_uri String,
    service LowCardinality(String),
    cnt UInt64,
    p50 AggregateFunction(quantile, UInt32),
    p95 AggregateFunction(quantile, UInt32),
    p99 AggregateFunction(quantile, UInt32)
) ENGINE = AggregatingMergeTree
ORDER BY (service, request_uri, window_start);

CREATE MATERIALIZED VIEW slow_requests_mv TO dws.slow_requests AS
SELECT
    toStartOfMinute(log_time) AS window_start,
    request_uri,
    service,
    count() AS cnt,
    quantileState(0.5)(latency_ms) AS p50,
    quantileState(0.95)(latency_ms) AS p95,
    quantileState(0.99)(latency_ms) AS p99
FROM ods.app_logs
WHERE status_code = 200
  AND log_time >= now() - INTERVAL 1 DAY
GROUP BY window_start, request_uri, service;

-- 查询 Top 10 慢接口
SELECT
    request_uri,
    sum(cnt) AS total_requests,
    round(quantileMerge(0.95)(p95), 2) AS p95_ms,
    round(quantileMerge(0.99)(p99), 2) AS p99_ms
FROM dws.slow_requests
WHERE window_start >= now() - INTERVAL 1 HOUR
  AND total_requests > 100
GROUP BY request_uri
ORDER BY p95_ms DESC
LIMIT 10;
```

### 11.2.5 链路追踪

```sql
-- 根据 trace_id 查询完整调用链
SELECT
    log_time,
    level,
    service,
    host,
    message,
    latency_ms
FROM ods.app_logs
WHERE trace_id = 'abc-123-xyz'
ORDER BY log_time;

-- 性能分析
WITH
    error_traces AS (
        SELECT DISTINCT trace_id
        FROM ods.app_logs
        WHERE log_time >= now() - INTERVAL 1 HOUR
          AND level = 'ERROR'
        LIMIT 100
    )
SELECT
    service,
    count() AS call_count,
    avg(latency_ms) AS avg_latency,
    quantile(0.99)(latency_ms) AS p99_latency
FROM ods.app_logs
WHERE trace_id IN (SELECT trace_id FROM error_traces)
GROUP BY service
ORDER BY call_count DESC;
```

### 11.2.6 用户行为日志分析

```sql
-- ========================================
-- 用户访问路径
-- ========================================
CREATE TABLE dws.user_sessions
(
    user_id UInt64,
    session_id String,
    first_event DateTime,
    last_event DateTime,
    duration_seconds UInt32,
    event_count UInt32,
    page_view_count UInt32,
    error_count UInt32
) ENGINE = SummingMergeTree
ORDER BY (user_id, session_id);

CREATE MATERIALIZED VIEW user_sessions_mv TO dws.user_sessions AS
WITH
    toStartOfHour(log_time) AS hour,
    concat(toString(user_id), '-', toString(hour)) AS session_id
SELECT
    user_id,
    session_id,
    min(log_time) AS first_event,
    max(log_time) AS last_event,
    dateDiff('second', min(log_time), max(log_time)) AS duration_seconds,
    count() AS event_count,
    countIf(event_type = 'page_view') AS page_view_count,
    countIf(level = 'ERROR') AS error_count
FROM ods.user_events
WHERE log_time >= now() - INTERVAL 7 DAY
GROUP BY user_id, session_id;

-- 用户活跃度
SELECT
    user_id,
    count(DISTINCT session_id) AS session_count,
    sum(event_count) AS total_events,
    sum(duration_seconds) AS total_time
FROM dws.user_sessions
WHERE first_event >= now() - INTERVAL 7 DAY
GROUP BY user_id
ORDER BY total_events DESC
LIMIT 100;
```

### 11.2.7 Nginx 访问日志专项分析

```sql
-- Nginx 日志表
CREATE TABLE ods.nginx_logs
(
    log_date Date,
    log_time DateTime,
    remote_addr String,
    remote_user String,
    method LowCardinality(String),
    request_uri String,
    http_version String,
    status_code UInt16,
    size_bytes UInt64,
    referer String,
    user_agent String,
    x_forwarded_for String,
    upstream_addr String,
    upstream_status UInt16,
    request_time Float32,
    upstream_response_time Float32
) ENGINE = MergeTree
PARTITION BY toYYYYMMDD(log_date)
ORDER BY (status_code, log_time)
TTL log_date + INTERVAL 90 DAY;

-- 状态码分布
SELECT
    status_code,
    count() AS cnt,
    round(cnt * 100.0 / sum(cnt) OVER (), 2) AS pct
FROM ods.nginx_logs
WHERE log_date = today()
GROUP BY status_code
ORDER BY cnt DESC;

-- 性能分析
SELECT
    request_uri,
    count() AS cnt,
    round(avg(request_time), 3) AS avg_time,
    round(quantile(0.95)(request_time), 3) AS p95_time,
    round(quantile(0.99)(request_time), 3) AS p99_time
FROM ods.nginx_logs
WHERE log_date = today()
  AND status_code = 200
  AND request_time > 0
GROUP BY request_uri
HAVING cnt > 100
ORDER BY avg_time DESC
LIMIT 20;

-- 异常请求（4xx/5xx）
SELECT
    status_code,
    request_uri,
    count() AS cnt,
    uniq(remote_addr) AS unique_ips,
    min(log_time) AS first_seen,
    max(log_time) AS last_seen
FROM ods.nginx_logs
WHERE log_date >= today() - 1
  AND status_code >= 400
GROUP BY status_code, request_uri
ORDER BY cnt DESC
LIMIT 50;
```

## 11.3 监控告警体系

```sql
-- ========================================
-- 1. 实时告警（错误率超阈值）
-- ========================================
SELECT
    service,
    count() AS error_count,
    uniq(trace_id) AS unique_traces,
    groupArray(DISTINCT message) AS sample_messages
FROM ods.app_logs
WHERE log_time >= now() - INTERVAL 5 MINUTE
  AND level = 'ERROR'
GROUP BY service
HAVING error_count > 10
ORDER BY error_count DESC;

-- ========================================
-- 2. 系统指标异常
-- ========================================
SELECT
    service,
    countIf(event_type = 'gc_pause' AND latency_ms > 1000) AS slow_gc,
    countIf(event_type = 'cpu_high' AND latency_ms > 80) AS cpu_high,
    countIf(event_type = 'memory_pressure' AND latency_ms > 90) AS memory_pressure
FROM ods.app_logs
WHERE log_time >= now() - INTERVAL 1 HOUR
GROUP BY service
ORDER BY slow_gc DESC;

-- ========================================
-- 3. 慢 SQL 日志
-- ========================================
CREATE TABLE ods.slow_sql
(
    log_date Date,
    log_time DateTime,
    service LowCardinality(String),
    db LowCardinality(String),
    sql String,
    duration_ms UInt32,
    rows UInt64,
    user String
) ENGINE = MergeTree
PARTITION BY toYYYYMMDD(log_date)
ORDER BY (db, log_time);

-- 慢 SQL Top 10
SELECT
    db,
    sql,
    count() AS cnt,
    avg(duration_ms) AS avg_ms,
    max(duration_ms) AS max_ms,
    uniq(user) AS unique_users
FROM ods.slow_sql
WHERE log_date >= today() - 7
GROUP BY db, sql
HAVING cnt > 10
ORDER BY avg_ms DESC
LIMIT 10;
```

## 11.4 业务监控大屏

```sql
-- ========================================
-- 整体服务健康度
-- ========================================
SELECT
    service,
    -- 1 分钟窗口
    countIf(level = 'ERROR' AND log_time >= now() - INTERVAL 1 MINUTE) AS errors_1m,
    countIf(level = 'WARN' AND log_time >= now() - INTERVAL 1 MINUTE) AS warns_1m,
    -- 5 分钟
    countIf(level = 'ERROR' AND log_time >= now() - INTERVAL 5 MINUTE) AS errors_5m,
    countIf(level = 'WARN' AND log_time >= now() - INTERVAL 5 MINUTE) AS warns_5m,
    -- 1 小时
    countIf(level = 'ERROR' AND log_time >= now() - INTERVAL 1 HOUR) AS errors_1h
FROM ods.app_logs
WHERE log_time >= now() - INTERVAL 1 HOUR
GROUP BY service
ORDER BY errors_1m DESC;

-- ========================================
-- 业务关键指标
-- ========================================
SELECT
    -- 订单相关
    countIf(event_type = 'order_created' AND log_time >= now() - INTERVAL 5 MINUTE) AS orders_5m,
    countIf(event_type = 'order_paid' AND log_time >= now() - INTERVAL 5 MINUTE) AS paid_5m,
    -- 转化率
    round(paid_5m * 100.0 / orders_5m, 2) AS conversion_pct,
    -- 金额
    sumIf(toFloat64OrZero(properties['amount']), event_type = 'order_paid' AND log_time >= now() - INTERVAL 5 MINUTE) AS gmv_5m
FROM ods.business_events
WHERE log_time >= now() - INTERVAL 5 MINUTE;
```
