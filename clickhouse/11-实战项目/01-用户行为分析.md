# 第 11 章：实战项目

## 项目一：用户行为分析平台

### 11.1.1 业务背景

构建一个支持**日增 100 亿事件**的用户行为分析平台，提供：
- 实时漏斗
- 用户留存
- 路径分析
- 行为画像
- A/B 测试效果分析

### 11.1.2 数据建模

```sql
-- ========================================
-- 1. 原始事件表
-- ========================================
CREATE TABLE ods.user_events
(
    event_id UInt64,
    event_time DateTime64(3),
    event_date Date MATERIALIZED toDate(event_time),
    user_id UInt64,
    session_id String,
    event_type LowCardinality(String),  -- 事件类型
    page_url String DEFAULT '',
    element_id String DEFAULT '',
    properties Map(String, String),     -- 扩展属性
    device LowCardinality(String),      -- ios/android/web
    app_version String DEFAULT '',
    INDEX idx_user user_id TYPE bloom_filter(0.01) GRANULARITY 4,
    INDEX idx_session session_id TYPE bloom_filter(0.01) GRANULARITY 4,
    INDEX idx_url page_url TYPE tokenbf_v1(25600, 3, 0) GRANULARITY 4
) ENGINE = MergeTree
PARTITION BY toYYYYMMDD(event_date)
ORDER BY (event_type, user_id, event_time)
TTL event_date + INTERVAL 90 DAY
SETTINGS index_granularity = 8192;

-- ========================================
-- 2. 用户维度表
-- ========================================
CREATE TABLE dwd.dim_user
(
    user_id UInt64,
    username String,
    register_date Date,
    register_channel LowCardinality(String),
    city LowCardinality(String),
    age_group LowCardinality(String),  -- 18-24, 25-30, ...
    gender LowCardinality(String),
    vip_level UInt8 DEFAULT 0,
    last_active_at DateTime
) ENGINE = ReplacingMergeTree(last_active_at)
ORDER BY user_id;

-- ========================================
-- 3. 会话聚合表
-- ========================================
CREATE TABLE dws.user_session_agg
(
    session_id String,
    user_id UInt64,
    session_start DateTime,
    session_end DateTime,
    session_date Date MATERIALIZED toDate(session_start),
    duration UInt32,  -- 秒
    event_count UInt32,
    page_view_count UInt32,
    click_count UInt32
) ENGINE = AggregatingMergeTree
PARTITION BY toYYYYMM(session_date)
ORDER BY (user_id, session_start);

-- 物化视图
CREATE MATERIALIZED VIEW user_session_mv TO dws.user_session_agg AS
SELECT
    session_id,
    user_id,
    min(event_time) AS session_start,
    max(event_time) AS session_end,
    dateDiff('second', min(event_time), max(event_time)) AS duration,
    count() AS event_count,
    countIf(event_type = 'page_view') AS page_view_count,
    countIf(event_type = 'click') AS click_count
FROM ods.user_events
GROUP BY session_id, user_id;
```

### 11.1.3 实时漏斗分析

```sql
-- ========================================
-- 漏斗表（每 5 分钟更新）
-- ========================================
CREATE TABLE dws.funnel_5min
(
    funnel_name LowCardinality(String),
    window_start DateTime,
    user_id UInt64,
    -- 是否完成各步骤
    step1 UInt8,
    step2 UInt8,
    step3 UInt8,
    step4 UInt8
) ENGINE = ReplacingMergeTree
PARTITION BY toYYYYMM(window_start)
ORDER BY (funnel_name, window_start, user_id);

CREATE MATERIALIZED VIEW funnel_5min_mv TO dws.funnel_5min AS
WITH
    toStartOfFiveMinute(event_time) AS window_start
SELECT
    'purchase_funnel' AS funnel_name,
    window_start,
    user_id,
    max(if(event_type = 'view_product', 1, 0)) AS step1,
    max(if(event_type = 'add_to_cart', 1, 0)) AS step2,
    max(if(event_type = 'checkout', 1, 0)) AS step3,
    max(if(event_type = 'pay_success', 1, 0)) AS step4
FROM ods.user_events
WHERE event_time >= now() - INTERVAL 7 DAY
GROUP BY window_start, user_id;

-- ========================================
-- 漏斗查询
-- ========================================
SELECT
    toStartOfHour(window_start) AS hour,
    -- 每一步的人数
    countIf(step1 = 1) AS step1_users,  -- 浏览商品
    countIf(step1 = 1 AND step2 = 1) AS step2_users,  -- 加购
    countIf(step1 = 1 AND step2 = 1 AND step3 = 1) AS step3_users,  -- 结算
    countIf(step1 = 1 AND step2 = 1 AND step3 = 1 AND step4 = 1) AS step4_users,  -- 支付成功
    -- 转化率
    round(countIf(step1 = 1 AND step2 = 1) * 100.0 / countIf(step1 = 1), 2) AS step1_to_step2,
    round(countIf(step1 = 1 AND step2 = 1 AND step3 = 1) * 100.0 / countIf(step1 = 1 AND step2 = 1), 2) AS step2_to_step3,
    round(countIf(step1 = 1 AND step2 = 1 AND step3 = 1 AND step4 = 1) * 100.0 / countIf(step1 = 1 AND step2 = 1 AND step3 = 1), 2) AS step3_to_step4
FROM dws.funnel_5min
WHERE window_start >= now() - INTERVAL 24 HOUR
GROUP BY hour
ORDER BY hour;
```

### 11.1.4 用户留存分析

```sql
-- ========================================
-- 留存计算
-- ========================================
WITH
    -- 新用户（首日活跃）
    new_users AS (
        SELECT
            user_id,
            min(event_date) AS first_active_date
        FROM ods.user_events
        WHERE event_date >= today() - 30
        GROUP BY user_id
    ),
    -- 后续活跃
    active_users AS (
        SELECT DISTINCT
            user_id,
            event_date
        FROM ods.user_events
        WHERE event_date >= today() - 30
    )
SELECT
    first_active_date AS cohort_date,
    count() AS cohort_size,
    -- 1 日留存
    round(countIf(
        user_id IN (SELECT user_id FROM active_users
                    WHERE event_date = first_active_date + 1)
    ) * 100.0 / count(), 2) AS d1_retention,
    -- 3 日留存
    round(countIf(
        user_id IN (SELECT user_id FROM active_users
                    WHERE event_date = first_active_date + 3)
    ) * 100.0 / count(), 2) AS d3_retention,
    -- 7 日留存
    round(countIf(
        user_id IN (SELECT user_id FROM active_users
                    WHERE event_date = first_active_date + 7)
    ) * 100.0 / count(), 2) AS d7_retention,
    -- 30 日留存
    round(countIf(
        user_id IN (SELECT user_id FROM active_users
                    WHERE event_date = first_active_date + 30)
    ) * 100.0 / count(), 2) AS d30_retention
FROM new_users
WHERE first_active_date BETWEEN today() - 30 AND today() - 1
GROUP BY first_active_date
ORDER BY first_active_date;
```

### 11.1.5 路径分析（桑基图）

```sql
-- 步骤1：提取路径
WITH
    user_paths AS (
        SELECT
            user_id,
            session_id,
            groupArray(event_type) AS path
        FROM (
            SELECT
                user_id,
                session_id,
                event_type,
                event_time
            FROM ods.user_events
            WHERE event_date = today()
            ORDER BY user_id, session_id, event_time
        )
        GROUP BY user_id, session_id
    ),
    -- 步骤2：两两相邻组合
    path_pairs AS (
        SELECT
            arrayJoin(arrayZip(
                arrayResize(path, length(path) - 1, ''),
                arraySlice(path, 2, length(path) - 1)
            )) AS pair
        FROM user_paths
    )
SELECT
    pair.1 AS from_step,
    pair.2 AS to_step,
    count() AS transition_count
FROM path_pairs
WHERE pair.1 != '' AND pair.2 != ''
GROUP BY from_step, to_step
ORDER BY transition_count DESC
LIMIT 50;
```

### 11.1.6 用户画像查询

```sql
-- RFM 分析
WITH
    user_metrics AS (
        SELECT
            user_id,
            max(event_time) AS last_active,
            count() AS frequency,
            sum(if(event_type = 'pay_success', toFloat64OrZero(properties['amount']), 0)) AS monetary
        FROM ods.user_events
        WHERE event_date >= today() - 90
        GROUP BY user_id
    )
SELECT
    user_id,
    -- R：最近一次消费距今天数
    dateDiff('day', last_active, now()) AS recency,
    -- F：消费频次
    frequency,
    -- M：消费金额
    monetary,
    -- RFM 分层
    multiIf(
        recency <= 7 AND frequency >= 10 AND monetary >= 1000, '高价值客户',
        recency <= 30 AND frequency >= 5, '忠实客户',
        recency <= 90, '一般客户',
        '流失客户'
    ) AS rfm_segment
FROM user_metrics
ORDER BY monetary DESC
LIMIT 1000;
```

```sql
-- ========================================
-- 原始日志表
-- ========================================
CREATE TABLE ods.app_logs
(
    log_date Date,
    log_time DateTime,
    level LowCardinality(String),
    service LowCardinality(String),
    host String,
    trace_id String,
    span_id String,
    message String,

    INDEX idx_uri request_uri TYPE tokenbf_v1(25600, 3, 0) GRANULARITY 4

PARTITION BY toYYYYMMDD(log_date)

-- Kafka 接入

(
    level String,
    service String,
    host String,
    trace_id String,
    message String,
    user_id UInt64 DEFAULT 0,
    request_uri String DEFAULT '',
    method String DEFAULT '',
    status_code UInt16 DEFAULT 0,
    latency_ms UInt32 DEFAULT 0,
    log_time DateTime DEFAULT now()
) ENGINE = Kafka
SETTINGS
    kafka_broker_list = 'kafka1:9092,kafka2:9092',
    kafka_topic_list = 'app_logs',
    kafka_group_name = 'log_consumer',
    kafka_format = 'JSONEachRow',
    kafka_handle_error_mode = 'stream',
    kafka_num_consumers = 4,
    kafka_thread_per_consumer = 2;

CREATE MATERIALIZED VIEW app_logs_mv TO ods.app_logs AS
SELECT
    toDate(log_time) AS log_date,
    log_time,
    level,
    service,
    host,
    trace_id,
    '' AS span_id,
    message,
    user_id,
    request_uri,
    method,
    status_code,
    latency_ms
FROM ods.app_logs_kafka;
```

### 11.2.3 错误告警查询

```sql
-- 实时错误率
SELECT
    toStartOfMinute(log_time) AS minute,
    service,
    countIf(level = 'ERROR') AS error_count,
    count() AS total_count,
    round(error_count * 100.0 / total_count, 2) AS error_rate
FROM ods.app_logs
WHERE log_time >= now() - INTERVAL 1 HOUR
  AND level IN ('ERROR', 'WARN', 'INFO')
GROUP BY minute, service
HAVING error_rate > 5
ORDER BY minute DESC;

-- 慢查询 Top 10
SELECT
    request_uri,
    count() AS cnt,
    avg(latency_ms) AS avg_ms,
    quantile(0.95)(latency_ms) AS p95_ms,
    max(latency_ms) AS max_ms
FROM ods.app_logs
WHERE log_time >= now() - INTERVAL 1 HOUR
  AND status_code = 200
  AND method = 'GET'
GROUP BY request_uri
ORDER BY avg_ms DESC
LIMIT 10;
```

### 11.2.4 链路追踪查询

```sql
-- 根据 trace_id 拉取完整调用链

    level,
    service,
    host,
    message,
    latency_ms
FROM ods.app_logs

-- 性能分析
SELECT
    service,
    count() AS call_count,
    avg(latency_ms) AS avg_latency,
    quantile(0.99)(latency_ms) AS p99_latency
FROM ods.app_logs
WHERE trace_id IN (
    SELECT trace_id
    FROM ods.app_logs
    WHERE log_time >= now() - INTERVAL 1 HOUR
      AND level = 'ERROR'
    LIMIT 100
)
GROUP BY service
ORDER BY call_count DESC;
```

### 11.3.1 业务背景

### 11.3.2 数据准备

```sql
-- 每分钟聚合
CREATE MATERIALIZED VIEW realtime_minute_mv
ENGINE = SummingMergeTree
PARTITION BY toYYYYMM(minute)
ORDER BY (metric, minute)
AS SELECT
    toStartOfMinute(event_time) AS minute,
    event_type AS metric,
    count() AS cnt
FROM ods.user_events
WHERE event_time >= now() - INTERVAL 1 DAY
GROUP BY minute, event_type;

-- 当日累计
CREATE TABLE realtime_daily
(
    metric LowCardinality(String),
    cnt UInt64
) ENGINE = SummingMergeTree
ORDER BY metric;

CREATE MATERIALIZED VIEW realtime_daily_mv TO realtime_daily AS
SELECT
    event_type AS metric,
    count() AS cnt
FROM ods.user_events
WHERE toDate(event_time) = today()
GROUP BY metric;
```

### 11.3.3 大屏 SQL

```sql
-- ========================================
-- 大屏 1：核心指标
-- ========================================
SELECT
    'DAU' AS metric,
    uniqExact(user_id) AS value
FROM ods.user_events
WHERE toDate(event_time) = today()
UNION ALL
SELECT '订单数', count()
FROM ods.user_events
WHERE toDate(event_time) = today() AND event_type = 'pay_success'
UNION ALL
SELECT 'GMV', sum(toFloat64OrZero(properties['amount']))
FROM ods.user_events
WHERE toDate(event_time) = today() AND event_type = 'pay_success';

-- ========================================
-- 大屏 2：趋势图
-- ========================================
SELECT
    minute,
    sumIf(cnt, metric = 'page_view') AS pv,
    sumIf(cnt, metric = 'click') AS click,
    sumIf(cnt, metric = 'pay_success') AS pay
FROM realtime_minute
WHERE minute >= now() - INTERVAL 2 HOUR
GROUP BY minute
ORDER BY minute;

-- ========================================
-- 大屏 3：实时排行
-- ========================================
SELECT
    properties['product_id'] AS product_id,
    count() AS sales
FROM ods.user_events
WHERE event_time >= now() - INTERVAL 1 HOUR
  AND event_type = 'pay_success'
GROUP BY product_id
ORDER BY sales DESC
LIMIT 10;

-- ========================================
-- 大屏 4：地域分布
-- ========================================
SELECT
    properties['city'] AS city,
    count() AS cnt
FROM ods.user_events
WHERE event_time >= now() - INTERVAL 5 MINUTE
  AND event_type = 'page_view'
GROUP BY city
ORDER BY cnt DESC
LIMIT 20;
```

## 项目四：A/B 测试效果分析

### 11.4.1 数据准备

```sql
-- A/B 实验配置
CREATE TABLE ods.ab_experiments
(
    experiment_id String,
    user_id UInt64,
    variant LowCardinality(String),  -- A / B / control
    assigned_at DateTime
) ENGINE = ReplacingMergeTree(assigned_at)
ORDER BY (experiment_id, user_id);

-- 业务数据关联
SELECT
    e.experiment_id,
    e.variant,
    count(DISTINCT e.user_id) AS users,
    countIf(ev.event_type = 'pay_success') AS conversions,
    sumIf(toFloat64OrZero(ev.properties['amount']), ev.event_type = 'pay_success') AS revenue
FROM ods.ab_experiments e
LEFT JOIN ods.user_events ev
    ON e.user_id = ev.user_id
    AND ev.event_time BETWEEN e.assigned_at AND e.assigned_at + INTERVAL 7 DAY
WHERE e.experiment_id = 'checkout_redesign_2024'
GROUP BY e.experiment_id, e.variant;
```

### 11.4.2 显著性检验

```sql
-- Z 检验（转化率）
WITH
    variant_stats AS (
        SELECT
            variant,
            count() AS users,
            countIf(event_type = 'pay_success') AS conversions
        FROM ods.ab_experiments e
        LEFT JOIN ods.user_events ev ON e.user_id = ev.user_id
            AND ev.event_time BETWEEN e.assigned_at AND e.assigned_at + INTERVAL 7 DAY
        WHERE e.experiment_id = 'checkout_redesign_2024'
        GROUP BY variant
    ),
    -- 计算转化率
    rates AS (
        SELECT
            variant,
            users,
            conversions,
            conversions / users AS conversion_rate
        FROM variant_stats
    )
SELECT
    -- A 组指标
    (SELECT conversion_rate FROM rates WHERE variant = 'A') AS a_rate,
    (SELECT conversion_rate FROM rates WHERE variant = 'B') AS b_rate,
    -- 提升幅度
    round(
        ((SELECT conversion_rate FROM rates WHERE variant = 'B') -
         (SELECT conversion_rate FROM rates WHERE variant = 'A'))
        * 100.0 / (SELECT conversion_rate FROM rates WHERE variant = 'A'),
    2) AS uplift_pct;
```

## 项目五：数据血缘与监控

### 11.5.1 慢查询监控看板

```sql
-- 慢查询 Top 20
SELECT
    formatDateTime(event_time, '%Y-%m-%d %H:%M:%S') AS time,
    user,
    formatReadableSize(memory_usage) AS memory,
    formatReadableSize(read_bytes) AS read_size,
    query_duration_ms AS duration_ms,
    read_rows,
    substring(query, 1, 200) AS query_preview
FROM system.query_log
WHERE type > 1
  AND query_duration_ms > 1000
  AND event_time >= now() - INTERVAL 1 DAY
ORDER BY query_duration_ms DESC
LIMIT 20;

-- 用户查询量
SELECT
    user,
    count() AS total_queries,
    countIf(query_duration_ms > 1000) AS slow_queries,
    round(avg(query_duration_ms), 2) AS avg_duration_ms,
    round(quantile(0.95)(query_duration_ms), 2) AS p95_duration_ms
FROM system.query_log
WHERE type > 1
  AND event_time >= now() - INTERVAL 1 DAY
GROUP BY user
ORDER BY total_queries DESC;
```
