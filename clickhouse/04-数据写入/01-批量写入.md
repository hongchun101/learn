# 第 4 章：数据写入与 ETL

## 4.1 数据写入模式全景

```
数据源 → ClickHouse
├── 离线批量（OLAP 场景）
│   ├── INSERT INTO VALUES
│   ├── INSERT FROM SELECT
│   ├── clickhouse-client < file.csv
│   ├── HTTP 接口 (TabSeparated/CSV/JSONEachRow)
│   └── 外部工具（DataX、Airbyte）
├── 实时流式
│   ├── Kafka Engine
│   ├── Materialized View + Kafka
│   ├── Buffer Engine
│   └── Kafka Connect ClickHouse Sink
├── CDC 同步
│   ├── Debezium + Kafka + ClickHouse
│   ├── Materialize CDC
│   └── 自研 ETL 工具
└── 联邦查询
    ├── MySQL/PostgreSQL 表函数
    ├── ODBC/JDBC
    └── S3/HDFS 外部存储
```

## 4.2 批量写入：速度对比

### 4.2.1 INSERT VALUES 性能测试

```sql
-- 准备目标表
CREATE TABLE perf_test
(
    id UInt64,
    name String,
    value Float64,
    ts DateTime
) ENGINE = MergeTree ORDER BY id;

-- 测试 1：单条 INSERT
INSERT INTO perf_test VALUES (1, 'a', 1.0, now());  -- ~50ms

-- 测试 2：批量 1000 条
INSERT INTO perf_test VALUES
    (2, 'b', 2.0, now()),
    (3, 'c', 3.0, now()),
    ...
    (1001, 'z', 1001.0, now());  -- ~200ms（5000 行/秒）

-- 测试 3：批量 10 万条
INSERT INTO perf_test VALUES
    ...  -- 10 万行
-- 耗时：~2s （5万 行/秒）

-- 测试 4：单条 INSERT 1000 次（❌ 极慢）
-- for i in 1..1000: INSERT VALUES (i, ...)
-- 耗时：~60s （17 行/秒，慢 3000 倍）
```

### 4.2.2 性能瓶颈分析

```sql
-- 单条慢的原因：
-- 1. 每次 INSERT 创建一个 part 文件
-- 2. Part 文件太多，merge 压力大
-- 3. 网络往返开销

-- 批量快的原因：
-- 1. 单次网络往返
-- 2. 一个 part 包含 10万 行
-- 3. 压缩效率高
```

## 4.3 写入性能优化

### 4.3.1 关键参数

```sql
-- 单查询最大内存
SET max_memory_usage = 10000000000;  -- 10GB

-- 异步插入（24.x+）
SET async_insert = 1;
SET wait_for_async_insert = 1;
SET async_insert_max_data_size = 1000000;  -- 1MB 触发
SET async_insert_busy_timeout_ms = 200;   -- 200ms 超时

-- 批量大小
SET max_insert_block_size = 1048576;  -- 1M 行

-- Parts 控制
SET max_parts_in_total = 100000;
SET parts_to_throw_insert = 300;
SET parts_to_delay_insert = 150;
```

### 4.3.2 最佳实践

```sql
-- 1. 每批 10万 - 100万 行
INSERT INTO target SELECT * FROM source LIMIT 1000000;

-- 2. 使用 TabSeparated 格式（最快）
-- clickhouse-client --format=TabSeparated < data.tsv

-- 3. 关闭同步到远端（如有副本）
SETTINGS distributed_foreground_insert = 0;

-- 4. 写入前排序（按 ORDER BY 键）
INSERT INTO target
SELECT * FROM source
ORDER BY sort_key_column;

-- 5. 使用临时表 + 批量插入
CREATE TABLE temp_target AS target;
INSERT INTO temp_target SELECT * FROM source;
-- 验证数据
SELECT count() FROM temp_target;
INSERT INTO target SELECT * FROM temp_target;
DROP TABLE temp_target;
```

## 4.4 异步插入

```sql
-- 启用异步插入
SETTINGS
  async_insert = 1,
  wait_for_async_insert = 0,  -- 不等待刷盘
  async_insert_max_data_size = 10485760  -- 10MB
```

**适用场景**：
- 大量小批量写入（如 1KB/次的日志）
- 客户端有 buffer（Kafka consumer）
- 不需要强一致性

**不适用**：
- 大批量写入（已是最优）
- 需要立即查询
- 强一致性要求

## 4.5 不同格式性能对比

```bash
# 准备测试数据（100 万行）
python3 -c "

for i in range(1000000):
    print(f'{i}\tname_{i}\t{random.random()}\t2024-01-01 00:00:00')

  --query "INSERT INTO perf_test FORMAT CSV" < /tmp/test.csv

  --query "INSERT INTO perf_test SELECT * FROM source FORMAT Native"

```

## 4.6 实战：Kafka 实时写入

### 4.6.1 Kafka 引擎表

```sql
-- 1. 创建 Kafka 消费表
CREATE TABLE events_kafka
(
    event_date Date,
    event_time DateTime,
    user_id UInt64,
    event_type LowCardinality(String),
    amount Decimal(10, 2)
) ENGINE = Kafka
SETTINGS
    kafka_broker_list = 'kafka1:9092,kafka2:9092,kafka3:9092',
    kafka_topic_list = 'user_events',
    kafka_group_name = 'clickhouse_consumer_1',
    kafka_format = 'JSONEachRow',
    kafka_num_consumers = 1,
    kafka_thread_per_consumer = 1,
    kafka_handle_error_mode = 'stream',
    kafka_max_block_size = 1048576;

-- 2. 创建目标 MergeTree 表
CREATE TABLE events_target
(
    event_date Date,
    event_time DateTime,
    user_id UInt64,
    event_type LowCardinality(String),
    amount Decimal(10, 2)
) ENGINE = MergeTree
PARTITION BY toYYYYMMDD(event_date)
ORDER BY (event_type, user_id, event_time);

-- 3. 创建物化视图（实时管道）
CREATE MATERIALIZED VIEW events_mv TO events_target AS
SELECT * FROM events_kafka;

-- 4. 查询数据
SELECT count() FROM events_target;

-- 5. 查看消费进度
SELECT *
FROM system.kafka_consumers
WHERE topic = 'user_events';
```

### 4.6.2 Kafka 写入性能调优

```sql
-- 高吞吐配置
SETTINGS
  kafka_num_consumers = 4,           -- 4 个消费者
  kafka_thread_per_consumer = 2,    -- 每消费者 2 线程
  -- 8 个并发消费 partition

  -- 批量大小
  kafka_max_block_size = 1048576,    -- 1MB
  kafka_poll_timeout_ms = 500,       -- 500ms 拉取一次

  -- 错误处理
  kafka_handle_error_mode = 'stream',
  kafka_commit_on_select = 1
```

## 4.7 Materialized View（物化视图）

### 4.7.1 普通物化视图

```sql
-- 实时聚合：统计每分钟每个事件类型
CREATE MATERIALIZED VIEW events_per_minute_mv
ENGINE = SummingMergeTree
PARTITION BY toYYYYMM(minute)
ORDER BY (event_type, minute)
AS SELECT
    toStartOfMinute(event_time) AS minute,
    event_type,
    count() AS cnt,
    uniq(user_id) AS uv
FROM events_kafka
GROUP BY minute, event_type;

-- 查询
SELECT * FROM events_per_minute_mv ORDER BY minute DESC LIMIT 10;
```

### 4.7.2 增量物化视图（Refreshable MV，24.x+）

```sql
-- 定期刷新的物化视图（替代 cron + INSERT）
CREATE MATERIALIZED VIEW daily_stats_mv
REFRESH EVERY 1 HOUR
AS SELECT
    toDate(event_time) AS date,
    event_type,
    count() AS cnt,
    uniqExact(user_id) AS uv
FROM events
GROUP BY date, event_type;

-- 手动触发
SYSTEM REFRESH VIEW daily_stats_mv;

-- 查看下次刷新时间
SELECT
    view,
    refresh_time,
    next_refresh_time,
    status
FROM system.view_refreshes;
```

## 4.8 数据转换 ETL

### 4.8.1 简单 ETL

```sql
-- 源表 → 目标表（数据清洗 + 转换）
INSERT INTO dwd.orders_clean
SELECT
    order_id,
    order_date,
    -- 清洗：去除前后空格
    trim(username) AS username,
    -- 转换：空值替换
    ifNull(phone, '') AS phone,
    -- 转换：金额单位
    amount / 100 AS amount_yuan,
    -- 提取：解析 JSON
    JSONExtractString(extra, 'channel') AS channel,
    -- 默认值
    now() AS etl_time
FROM ods.orders_raw
WHERE order_date >= '2024-01-01';
```

### 4.8.2 复杂 ETL：维度表构建

```sql
-- 用户维度表（每日全量）
CREATE TABLE dwd.dim_user
(
    user_id UInt64,
    username String,
    register_date Date,
    -- SCD Type 2：缓慢变化维
    effective_date Date,
    expire_date Date DEFAULT '9999-12-31',
    is_current UInt8 DEFAULT 1
) ENGINE = ReplacingMergeTree
ORDER BY (user_id, effective_date);

-- ETL 流程
INSERT INTO dwd.dim_user
SELECT
    user_id,
    username,
    register_date,
    today() AS effective_date,
    '9999-12-31' AS expire_date,
    1 AS is_current
FROM ods.user_changes
WHERE update_date = today();
```

## 4.9 实战：从 MySQL 同步数据

### 4.9.1 使用 MySQL 表引擎（不推荐生产）

```sql
-- 直接查询 MySQL
CREATE TABLE mysql_table
(
    id UInt64,
    name String,
    age UInt32
) ENGINE = MySQL('mysql_host:3306', 'db', 'table', 'user', 'password');

SELECT * FROM mysql_table LIMIT 100;
-- 性能：取决于 MySQL
```

### 4.9.2 使用 MaterializedMySQL（实时同步 CDC）

```sql
-- 1. 开启实验功能
SET allow_experimental_database_materialized_mysql = 1;

-- 2. 创建物化数据库
CREATE DATABASE mysql_sync
ENGINE = MaterializedMySQL('mysql_host:3306', 'source_db', 'user', 'password');

-- 3. 自动同步所有表
SHOW TABLES FROM mysql_sync;
SELECT * FROM mysql_sync.users LIMIT 10;

-- 4. 查看同步状态
SELECT
    database,
    table,
    is_readonly,
    total_rows
FROM system.tables
WHERE database = 'mysql_sync';
```
