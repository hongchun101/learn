# 第 2 章 续：数据类型详解与最佳实践

## 2.9 数据类型全景

```
ClickHouse 数据类型
├── 数值类型
│   ├── 整数：Int8/16/32/64, UInt8/16/32/64
│   ├── 浮点：Float32/64
│   └── 定点：Decimal32/64/128/256
├── 字符串类型
│   ├── String（变长）
│   ├── FixedString(N)（定长）
│   └── LowCardinality(T)（低基数包装器）
├── 时间类型
│   ├── Date（2 字节）
│   ├── Date32（4 字节）
│   ├── DateTime（4 字节）
│   └── DateTime64(N)（8 字节，毫秒/微秒/纳秒）
├── 复合类型
│   ├── Array(T)
│   ├── Tuple(T1, T2, ...)
│   ├── Map(K, V)
│   └── Nested(T1, T2, ...)
├── 特殊类型
│   ├── Nullable(T)
│   ├── IPv4/IPv6
│   ├── UUID
│   ├── Enum8/16
│   └── JSON（25.x 稳定）
└── 域类型
    └── 地理信息：Point, Ring, Polygon, MultiPolygon
```

## 2.10 类型选择决策树

```
需要存储什么数据？
│
├── 数字 ──────────→ 是整数？ ────→ 有负数？ ──┐
│     │            │              │ No        │ Yes
│     │            │              ↓           ↓
│     │            │           UInt8-64     Int8-64
│     │            │            │              │
│     │            │            ├── 是 ID？ ──→ UInt64
│     │            │            │
│     │            │            └── 业务数字 ─→ 看范围
│     │            │
│     │            └── 是小数？ ──→ 是金额？ ──┐
│     │                          │ Yes       │ No
│     │                          ↓           ↓
│     │                       Decimal       Float32/64
│     │
├── 字符串 ──────→ 是枚举？ ────→ 值数量 < 1万？ ──┐
│     │            │ Yes              │ Yes       │ No
│     │            │                  ↓           ↓
│     │            │        LowCardinality(String)   String
│     │            │
│     │            └── 否 ──→ 是定长？ ──┐
│     │                       │ Yes    │ No
│     │                       ↓        ↓
│     │                FixedString(N)  String
│     │
├── 时间 ──────────→ 仅日期？ ──→ Date（2字节）
│     │              │
│     │              └── 含时间？ ──→ 精度？
│     │                            ├ 秒级：DateTime
│     │                            ├ 毫秒：DateTime64(3)
│     │                            └ 微秒：DateTime64(6)
│     │
└── 复合 ──────────→ 看业务
                  ├ 标签/多值：Array(String)
                  ├ 异构字段：Map(String, String)
                  └ 嵌套：Nested(...)
```

## 2.11 实战：业务表类型选择

### 案例 1：用户画像表

```sql
CREATE TABLE dwd.user_profile
(
    -- 主键
    user_id        UInt64,
    
    -- 基础信息（字符串）
    username       String,
    phone          FixedString(11),  -- 定长 11 位
    
    -- 枚举（低基数）
    gender         LowCardinality(String),  -- M/F/U
    user_level     LowCardinality(String),  -- bronze/silver/gold
    city           LowCardinality(String),  -- 中国 300+ 城市
    
    -- 数值
    age            UInt8,                    -- 0-255
    balance        Decimal(18, 2),           -- 金额
    vip_score      UInt32,                   -- 0-100
    
    -- 标签（数组）
    interests      Array(LowCardinality(String)),
    
    -- 时间
    register_date  Date,
    register_time  DateTime,
    last_active_at DateTime,
    
    -- 扩展字段
    extra          Map(String, String)
)
ENGINE = MergeTree
ORDER BY user_id
SETTINGS index_granularity = 8192;
```

### 案例 2：物联网时序数据

```sql
CREATE TABLE dwd.device_metrics
(
    -- 设备 ID
    device_id      UInt32,
    sensor_id      UInt8,
    
    -- 时间（高基数）
    ts             DateTime64(3),  -- 毫秒精度
    metric_date    Date MATERIALIZED toDate(ts),  -- 物化列
    
    -- 指标（浮点）
    temperature    Float32,
    humidity       Float32,
    pressure       Float32,
    
    -- 状态
    status         UInt8,          -- 0=正常, 1=警告, 2=故障
    battery_level  UInt8           -- 0-100
)
ENGINE = MergeTree
PARTITION BY toYYYYMM(metric_date)
ORDER BY (device_id, sensor_id, ts)
TTL metric_date + INTERVAL 1 YEAR  -- 自动 1 年后删除
SETTINGS index_granularity = 8192;
```

### 案例 3：日志分析表

```sql
CREATE TABLE dwd.app_logs
(
    -- 时间
    log_date       Date,
    log_time       DateTime,
    
    -- 维度
    level          LowCardinality(String),  -- INFO/WARN/ERROR
    service        LowCardinality(String),  -- 50+ 微服务
    host           String,                  -- 容器 IP
    
    -- 消息
    trace_id       String,
    message        String,
    
    -- 结构化字段
    user_id        UInt64 DEFAULT 0,
    request_uri    String DEFAULT '',
    status_code    UInt16 DEFAULT 0,
    latency_ms     UInt32 DEFAULT 0,
    
    -- 索引
    INDEX idx_trace trace_id TYPE bloom_filter(0.01) GRANULARITY 4,
    INDEX idx_uri request_uri TYPE tokenbf_v1(25600, 3, 0) GRANULARITY 4
)
ENGINE = MergeTree
PARTITION BY toYYYYMMDD(log_date)
ORDER BY (level, service, log_time)
TTL log_date + INTERVAL 30 DAY;
```

## 2.12 性能对比测试

```sql
-- 测试 1：LowCardinality vs String（10 万行）
CREATE TABLE test.lc_test (s String) ENGINE = MergeTree ORDER BY tuple();
CREATE TABLE test.lc_lc (s LowCardinality(String)) ENGINE = MergeTree ORDER BY tuple();

INSERT INTO test.lc_test SELECT 'value_' || (number % 100) FROM numbers(1000000);
INSERT INTO test.lc_lc SELECT 'value_' || (number % 100) FROM numbers(1000000);

-- 存储对比
SELECT
    table,
    formatReadableSize(sum(bytes_on_disk)) AS size,
    formatReadableSize(sum(data_uncompressed_bytes)) AS raw
FROM system.parts
WHERE database = 'test' AND active
GROUP BY table;

-- 预期：
-- lc_test: 50MB raw, 15MB compressed
-- lc_lc:  15MB raw, 3MB compressed  （节省 5 倍）

-- 查询性能
SELECT count(), uniqExact(s) FROM test.lc_test;  -- 0.05s
SELECT count(), uniqExact(s) FROM test.lc_lc;    -- 0.02s （快 2-3 倍）
```

## 2.13 类型转换与函数

```sql
-- 显式转换
SELECT toInt64('123');           -- 123
SELECT toString(123);            -- '123'
SELECT toDate('2024-01-01');     -- 2024-01-01
SELECT toDateTime('2024-01-01 00:00:00');

-- 安全转换（无法转换返回 0）
SELECT toInt64OrZero('abc');     -- 0
SELECT toInt64OrNull('abc');     -- NULL

-- 类型推断
SELECT toTypeName(1);            -- UInt8
SELECT toTypeName(toInt64(1));   -- Int64
```

## 2.14 避坑指南

### 坑 1：Decimal 运算精度

```sql
-- ❌ 错误：精度丢失
SELECT toDecimal64(1.1, 2) * toDecimal64(2.2, 2);  -- 2.42 (对)
SELECT 1.1 * 2.2;                                   -- 2.42 (浮点)

-- ✅ 正确：所有运算都用 Decimal
SELECT
    toDecimal64(1.1, 2) AS a,
    toDecimal64(2.2, 2) AS b,
    a * b;  -- 严格按 Decimal 规则
```

### 坑 2：Nullable 性能损耗

```sql
-- 测试：100 万行聚合
CREATE TABLE t_nullable (x Nullable(UInt32)) ENGINE = MergeTree ORDER BY tuple();
CREATE TABLE t_default (x UInt32 DEFAULT 0) ENGINE = MergeTree ORDER BY tuple();

INSERT INTO t_nullable SELECT number % 1000 FROM numbers(1000000);
INSERT INTO t_default SELECT number % 1000 FROM numbers(1000000);

-- 查询
SELECT avg(x) FROM t_nullable;  -- 0.15s
SELECT avg(x) FROM t_default;   -- 0.08s （快 2 倍）
```

### 坑 3：Array 索引性能

```sql
-- 创建数组列
CREATE TABLE t_array
(
    id UInt64,
    tags Array(String)
) ENGINE = MergeTree ORDER BY id;

-- ❌ 全表扫描
SELECT * FROM t_array WHERE 'premium' IN tags;
-- 性能差

-- ✅ 优化：使用 arrayExists 等函数
SELECT * FROM t_array WHERE arrayExists(x -> x = 'premium', tags);
-- 或构建扁平表（推荐）

-- 终极方案：单独建标签表
CREATE TABLE t_tags
(
    id UInt64,
    tag String
) ENGINE = MergeTree ORDER BY (tag, id);
```
