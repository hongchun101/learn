# 第 5 章：查询优化

## 5.1 查询执行流程

```
SQL 查询
  ↓
SQL 解析器 (Parser)  → AST
  ↓
分析器 (Analyzer)    → 查询计划树
  ↓
优化器 (Optimizer)   → 应用 RBO/CBO 规则
  ↓
执行器 (Executor)    → 并行执行 pipeline
  ↓
结果合并 → 返回
```

### 5.1.1 EXPLAIN 查看执行计划

```sql
-- 1. 普通 EXPLAIN
EXPLAIN
SELECT count()
FROM events
WHERE event_date = '2024-01-01'
  AND user_id = 1001;

-- 输出：
-- Expression ((Projection + Before ORDER BY))
--   Aggregating
--     Expression (Before GROUP BY)
--       Filter (WHERE)
--         ReadFromMergeTree (default.events)

-- 2. EXPLAIN AST
EXPLAIN AST
SELECT count() FROM events WHERE event_date = '2024-01-01';

-- 3. EXPLAIN SYNTAX
EXPLAIN SYNTAX
SELECT count() FROM events WHERE event_date = '2024-01-01';

-- 4. EXPLAIN PIPELINE（最详细）
EXPLAIN PIPELINE
SELECT count()
FROM events
WHERE event_date = '2024-01-01';

-- 5. EXPLAIN ESTIMATE（估算）
EXPLAIN ESTIMATE
SELECT count() FROM events WHERE event_date = '2024-01-01';
-- 返回：读取多少行、多少字节
```

## 5.2 主键索引（稀疏索引）

### 5.2.1 原理

```
数据存储（按 ORDER BY 排序）:
[user_1, event_a, time_1] [user_1, event_a, time_2] ... [user_2, event_b, time_1] ...

每个 8192 行打一个 mark，记录 user_id 的最大值

索引（primary.idx）:
Mark 0: user_id <= 100
Mark 1: user_id <= 200
Mark 2: user_id <= 300
...

查询 WHERE user_id = 150:
1. 找到 Mark 1
2. 读取 Mark 1 对应的 8192 行
3. 在内存中精确定位 user_id = 150
```

### 5.2.2 索引生效规则

```sql
-- 假设 ORDER BY (a, b, c)

-- ✅ 走索引：最左前缀
WHERE a = 1
WHERE a = 1 AND b = 2
WHERE a = 1 AND b = 2 AND c = 3

-- ✅ 部分走索引
WHERE a = 1 AND c = 3  -- 只用 a，c 全扫描

-- ❌ 不走索引
WHERE b = 2
WHERE c = 3
WHERE a > 1 AND b = 2  -- 范围查询后 b 无法用索引
```

### 5.2.3 实战：主键设计

```sql
-- 场景：用户行为分析
-- 高频查询：WHERE event_type = ? AND user_id = ? AND event_time > ?

-- ✅ 正确
CREATE TABLE events
(
    event_time DateTime,
    user_id UInt64,
    event_type LowCardinality(String),
    amount Decimal(10, 2)
) ENGINE = MergeTree
ORDER BY (event_type, user_id, event_time);
-- 前缀：event_type = 'click' → 大幅过滤
-- 再：user_id = 1001 → 精准定位
-- 后：event_time > '2024-01-01' → 范围扫描

-- ❌ 错误
ORDER BY (user_id, event_type, event_time);
-- user_id 基数太高，索引效果差
```

## 5.3 二级索引（跳数索引）

### 5.3.1 类型

```sql
-- 1. minmax：范围查询
INDEX idx_amount amount TYPE minmax GRANULARITY 4

-- 2. set：等值查询、IN
INDEX idx_event_type event_type TYPE set(100) GRANULARITY 4

-- 3. bloom_filter：高基数等值
INDEX idx_user_id user_id TYPE bloom_filter(0.01) GRANULARITY 4

-- 4. tokenbf_v1：字符串分词
INDEX idx_url url TYPE tokenbf_v1(25600, 3, 0) GRANULARITY 4

-- 5. ngrambf_v1：字符串 N-Gram
INDEX idx_message message TYPE ngrambf_v1(4, 25600, 3, 0) GRANULARITY 4
```

### 5.3.2 实战：日志分析

```sql
CREATE TABLE app_logs
(
    log_date Date,
    log_time DateTime,
    level LowCardinality(String),
    service LowCardinality(String),
    trace_id String,
    message String,
    -- 二级索引
    INDEX idx_trace trace_id TYPE bloom_filter(0.01) GRANULARITY 4,
    INDEX idx_message message TYPE tokenbf_v1(25600, 3, 0) GRANULARITY 4
) ENGINE = MergeTree
PARTITION BY toYYYYMMDD(log_date)
ORDER BY (level, service, log_time);

-- 查询：trace_id 精准查找
SELECT * FROM app_logs
WHERE trace_id = 'abc-123-xyz';
-- Bloom filter 过滤 99% 的 block

-- 查询：message 关键字
SELECT * FROM app_logs
WHERE message LIKE '%error%';
-- tokenbf_v1 加速
```

### 5.3.3 性能对比

```sql
-- 测试：1 亿行表，按 trace_id 查询
-- 无索引：5s
-- bloom_filter：0.2s （25 倍提升）
```

## 5.4 Projection（投影）

### 5.4.1 概念

Projection 是 ClickHouse 21.x 引入的特性，**相当于表内嵌的物化视图**。

```sql
-- 主表 ORDER BY a
-- 查询常用：WHERE b = 1 GROUP BY c
-- 解决方案：Projection 自动维护 b 排序的副本

CREATE TABLE events
(
    event_time DateTime,
    user_id UInt64,
    event_type String,
    amount Decimal(10, 2)
) ENGINE = MergeTree
ORDER BY (user_id, event_time);

-- 添加 projection
ALTER TABLE events
ADD PROJECTION proj_by_type
(
    SELECT *
    ORDER BY (event_type, event_time)
);

-- 物化数据
ALTER TABLE events MATERIALIZE PROJECTION proj_by_type;

-- 查询：自动选择最优 projection
SELECT count()
FROM events
WHERE event_type = 'click';
-- 自动使用 proj_by_type
```

### 5.4.2 实战

```sql
-- 场景：同一张表支持多种查询模式

CREATE TABLE orders
(
    order_id UInt64,
    user_id UInt64,
    merchant_id UInt32,
    order_date Date,
    amount Decimal(10, 2)
) ENGINE = MergeTree
ORDER BY (user_id, order_date);

-- 模式 1：按用户查
-- ORDER BY (user_id, order_date) ✅

-- 模式 2：按商家查
ALTER TABLE orders
ADD PROJECTION proj_by_merchant
(
    SELECT * ORDER BY (merchant_id, order_date)
);

-- 模式 3：按金额排
ALTER TABLE orders
ADD PROJECTION proj_by_amount
(
    SELECT * ORDER BY (amount, order_date)
);

-- 物化所有
ALTER TABLE orders MATERIALIZE PROJECTION proj_by_merchant;
ALTER TABLE orders MATERIALIZE PROJECTION proj_by_amount;

-- 验证：自动使用最优 projection
EXPLAIN SELECT * FROM orders WHERE merchant_id = 1001;
-- Projection: proj_by_merchant
```

## 5.5 PREWHERE 优化

### 5.5.1 原理

```
普通 WHERE：
1. 读取所有列到内存
2. 执行 WHERE 过滤
3. 输出结果

PREWHERE：
1. 读取 WHERE 条件中的列
2. 过滤
3. 补齐其他列（仅对未过滤掉的行）
```

### 5.5.2 实战对比

```sql
-- 1 亿行表，包含 100 列（总大小 100GB）
-- 只关心 event_date = '2024-01-01' 的 1% 数据

-- 普通 WHERE
SELECT count()
FROM events
WHERE event_date = '2024-01-01' AND user_id > 1000;
-- 读取：100GB（全部列），耗时 5s

-- PREWHERE
SELECT count()
FROM events
PREWHERE event_date = '2024-01-01'  -- 先用这一列过滤
WHERE user_id > 1000;
-- 读取：1GB（只剩 1%），耗时 0.5s （10 倍提升）

-- ✅ 推荐：让 ClickHouse 自动判断
-- enable_optimize_predicate_expression = 1（默认开启）
```

## 5.6 采样查询

```sql
-- SAMPLE 子句：基于哈希的近似查询

-- 创建表时指定采样键
CREATE TABLE events
(
    event_date Date,
    user_id UInt64,
    ...
) ENGINE = MergeTree
PARTITION BY toYYYYMM(event_date)
ORDER BY (user_id, event_date)
SAMPLE BY user_id;  -- 必须整数列

-- 10% 采样
SELECT count() FROM events SAMPLE 0.1;

-- 100万行采样
SELECT count() FROM events SAMPLE 1000000;

-- 1/10 采样
SELECT count() FROM events SAMPLE 1/10;

-- 采样 + WHERE
SELECT
    event_type,
    count() AS cnt_sample,
    count() * 10 AS cnt_estimated  -- 乘以 10 得到全表估算
FROM events SAMPLE 0.1
WHERE event_date = '2024-01-01'
GROUP BY event_type;
```

## 5.7 聚合优化

### 5.7.1 -State 函数 + 物化视图

```sql
-- 场景：每分钟统计 PV/UV

-- 1. 原始数据：100 万行/分钟
-- 2. 直接聚合：每次扫描 100 万行，10s
-- 3. 状态聚合：只扫描聚合后数据，0.01s

-- 第一步：建物化视图
CREATE MATERIALIZED VIEW events_minute_state_mv
ENGINE = AggregatingMergeTree
PARTITION BY toYYYYMM(minute)
ORDER BY (event_type, minute)
AS SELECT
    toStartOfMinute(event_time) AS minute,
    event_type,
    uniqState(user_id) AS uv_state,
    sumState(amount) AS amount_state,
    countState() AS cnt_state
FROM events
GROUP BY minute, event_type;

-- 第二步：查询合并状态
SELECT
    minute,
    event_type,
    uniqMerge(uv_state) AS uv,
    sumMerge(amount_state) AS amount,
    countMerge(cnt_state) AS cnt
FROM events_minute_state_mv
WHERE minute >= now() - INTERVAL 1 HOUR
GROUP BY minute, event_type
ORDER BY minute DESC;
-- 耗时：0.01s （1000 倍提升）
```

### 5.7.2 GROUP BY 优化

```sql
-- 1. 避免超大基数 GROUP BY
-- ❌ GROUP BY user_id, event_time（基数 = 总行数）
-- ✅ GROUP BY toStartOfHour(event_time), user_id（基数减少 3600 倍）

-- 2. 用 GROUP BY 替代 DISTINCT
-- ❌ SELECT DISTINCT user_id FROM events;
-- ✅ SELECT user_id FROM events GROUP BY user_id;  -- 略快

-- 3. 预聚合
CREATE TABLE events_hourly
(
    hour DateTime,
    event_type LowCardinality(String),
    cnt UInt64,
    uv AggregateFunction(uniq, UInt64)
) ENGINE = AggregatingMergeTree
ORDER BY (event_type, hour);

CREATE MATERIALIZED VIEW events_hourly_mv TO events_hourly AS
SELECT
    toStartOfHour(event_time) AS hour,
    event_type,
    count() AS cnt,
    uniqState(user_id) AS uv
FROM events
GROUP BY hour, event_type;
```

## 5.8 JOIN 优化

### 5.8.1 字典 JOIN（最佳）

```sql
-- 场景：100 亿行订单 JOIN 100 万行用户
-- 普通 JOIN：分钟级
-- 字典 JOIN：秒级

-- 1. 创建字典源表
CREATE TABLE dwd.users_dict_src
(
    user_id UInt64,
    city String,
    vip_level UInt8
) ENGINE = MergeTree ORDER BY user_id;

-- 2. 创建字典
CREATE DICTIONARY users_dict
(
    user_id UInt64,
    city String,
    vip_level UInt8
)
PRIMARY KEY user_id
SOURCE(CLICKHOUSE(DB 'dwd' TABLE 'users_dict_src'))
LIFETIME(MIN 300 MAX 600)
LAYOUT(HASHED());

-- 3. 使用字典
SELECT
    order_id,
    user_id,
    dictGet('users_dict', 'city', user_id) AS city,
    dictGet('users_dict', 'vip_level', user_id) AS vip_level
FROM orders;
-- 性能：20-30 倍提升
```

### 5.8.2 JOIN 算法选择

```sql
-- 1. hash join（默认）
SETTINGS join_algorithm = 'hash';

-- 2. parallel_hash（多线程）
SETTINGS join_algorithm = 'parallel_hash';

-- 3. grace hash（适合大表）
SETTINGS join_algorithm = 'grace_hash';

-- 4. full_sorting_merge（适合大表 JOIN 大表）
SETTINGS join_algorithm = 'full_sorting_merge';

-- 5. 强制内存
SETTINGS max_bytes_in_join = 10000000000;  -- 10GB
```

## 5.9 性能监控

### 5.9.1 query_log

```sql
-- 查看慢查询
SELECT
    query_duration_ms,
    query,
    user,
    type,
    read_rows,
    read_bytes,
    result_rows,
    formatReadableSize(read_bytes) AS read_size,
    formatReadableSize(memory_usage) AS mem
FROM system.query_log
WHERE type > 1
  AND query_duration_ms > 1000
  AND event_time > now() - INTERVAL 1 HOUR
ORDER BY query_duration_ms DESC
LIMIT 20;

-- 查看具体查询的执行信息
SELECT *
FROM system.query_log
WHERE query_id = 'xxx'
ORDER BY event_time;
```
